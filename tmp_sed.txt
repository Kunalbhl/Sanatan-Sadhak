/fun playGlobalAudio/i \
    private fun getSoundName(type: Int): String {\
        return when(type) {\
            1 -> "temple_bell"\
            2 -> "om_ambient"\
            3 -> "flute_ambient"\
            else -> "om_ambient"\
        }\
    }\

/fun playAmbientSound/i \
    fun getSoundName(index: Int): String {\
        return when (index) {\
            1 -> "temple_bell"\
            2 -> "om_ambient"\
            3 -> "flute_ambient"\
            4 -> "gayatri_mantra"\
            5 -> "shiva_tandav"\
            6 -> "maha_mrityunjaya"\
            7 -> "vishnu_sahasranamam"\
            else -> "om_ambient"\
        }\
    }\

/setContent {/i \
    override fun onDestroy() {\
        super.onDestroy()\
        com.example.ui.components.AudioEngine.releaseAll()\
    }\
